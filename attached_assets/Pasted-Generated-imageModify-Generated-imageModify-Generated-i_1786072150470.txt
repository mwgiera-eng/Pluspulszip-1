// HexbinHeatLayer.kt — Core heatmap rendering using Mapbox Style API

class HexbinHeatLayer(
    private val mapView: MapView,
    private val styleId: String = "pluspuls-dark-v1"
) {
    private var geoJsonSource: GeoJsonSource? = null
    
    companion object {
        const val SOURCE_ID = "demand-hex-source"
        const val LAYER_ID = "demand-hex-layer"
        const val LAYER_OUTLINE_ID = "demand-hex-outline"
        
        // Kraków center point for hex grid generation
        val KRAKOW_CENTER = Point.fromLngLat(19.9450, 50.0647)
        const val HEX_RADIUS_METERS = 300.0  // ~300m hexagons
        const val GRID_ROWS = 15
        const val GRID_COLS = 12
    }

    /**
     * Generate hexagonal grid covering Kraków city area.
     * Uses offset coordinate system for honeycomb tessellation.
     */
    fun generateHexGrid(
        demandData: List<ZoneDemand>
    ): FeatureCollection {
        val features = mutableListOf<Feature>()
        
        for (row in 0 until GRID_ROWS) {
            for (col in 0 until GRID_COLS) {
                // Offset coordinates for honeycomb pattern
                val offsetX = if (row % 2 == 0) 0.0 else HEX_RADIUS_METERS * 1.5
                val centerLng = KRAKOW_CENTER.longitude() + 
                    (col * HEX_RADIUS_METERS * 3 + offsetX) / 111320.0
                val centerLat = KRAKOW_CENTER.latitude() + 
                    (row * HEX_RADIUS_METERS * Math.sqrt(3.0)) / 110574.0
                
                // Find nearest demand zone for this hex
                val nearestDemand = demandData.minByOrNull { demand ->
                    haversineDistance(centerLat, centerLng, demand.lat, demand.lng)
                } ?: continue
                
                // Create hex polygon vertices
                val hexPolygon = createHexagonVertices(centerLng, centerLat, HEX_RADIUS_METERS)
                
                val feature = Feature.fromGeometry(Polygon.fromLngLats(listOf(hexPolygon)))
                feature.addBooleanProperty("isHot", nearestDemand.score > 70)
                feature.addNumberProperty("demandScore", nearestDemand.score)
                feature.addNumberProperty("surgeMultiplier", nearestDemand.surge)
                feature.addStringProperty("zoneName", nearestDemand.name)
                feature.addNumberProperty("confidence", nearestDemand.confidence)
                
                features.add(feature)
            }
        }
        
        return FeatureCollection.fromFeatures(features)
    }
    
    /**
     * Apply the hex grid to Mapbox map as a styled fill layer
     */
    fun applyToMap(mapboxMap: MapboxMap) {
        mapboxMap.getStyle { style ->
            // Add GeoJSON source
            geoJsonSource = GeoJsonSource(
                Builder(SOURCE_ID, generateHexGrid(emptyList()))
                    .maxzoom(16)
                    .tolerance(0.375f)
                    .buffer(64)
            )
            style.addSource(geoJsonSource!!)
            
            // Fill layer with interpolated colors based on demand score
            val fillLayer = FillLayer(LAYER_ID, SOURCE_ID)
                .withProperties(
                    fillColor(
                        interpolate(
                            expression { literal("demandScore") },
                            stop(0, rgba(220, 252, 231, 0.4)),   // #dcfce7 40%
                            stop(35, rgba(132, 204, 22, 0.5)),     // #84cc16 50%
                            stop(60, rgba(34, 197, 94, 0.6)),      // #22c55e 60%
                            stop(80, rgba(21, 128, 61, 0.85)),     // #15803d 85%
                            stop(95, rgba(220, 38, 38, 0.9))       // #dc2626 90%
                        )
                    ),
                    fillOpacity(
                        interpolate(
                            expression { literal("confidence") },
                            stop(0, 0.3),
                            stop(0.5, 0.5),
                            stop(1.0, 0.9)
                        )
                    )
                )
                .withFilter(ne(get("zoneName"), literal("")))
            
            // Outline layer for hex borders
            val outlineLayer = LineLayer(LAYER_OUTLINE_ID, SOURCE_ID)
                .withProperties(
                    lineColor(rgba(30, 58, 47, 0.6)),  // #1e3a2f
                    lineWidth(0.8f),
                    lineOpacity(0.7f)
                )
            
            style.addLayerBelow(fillLayer, "settlement-label")
            style.addLayerBelow(outlineLayer, LAYER_ID)
        }
    }
    
    /**
     * Update heat data without recreating layers
     */
    fun updateDemandData(demandData: List<ZoneDemand>) {
        geoJsonSource?.featureCollection(generateHexGrid(demandData))
    }
    
    /**
     * Animate transition when time slider changes
     */
    fun animateTimeTransition(
        newData: List<ZoneDemand>,
        durationMs: Long = 400
    ) {
        // Mapbox handles property transitions automatically
        // when source data updates — configure transition:
        mapView.getMapboxMap().getStyle { style ->
            style.transitionStyleOptions(
                TransitionOptions.Builder()
                    .duration(durationMs)
                    .delay(0)
                    .enablePlacementTransitions(true)
                    .build()
            )
            geoJsonSource?.featureCollection(generateHexGrid(newData))
        }
    }
    
    private fun createHexagonVertices(
        centerLng: Double,
        centerLat: Double,
        radiusMeters: Double
    ): List<Point> {
        val vertices = mutableListOf<Point>()
        val latPerMeter = 1.0 / 110574.0
        val lngPerMeter = 1.0 / (111320.0 * Math.cos(Math.toRadians(centerLat)))
        
        for (i in 0 until 6) {
            val angle = Math.PI / 3.0 * i + Math.PI / 6.0
            val dx = radiusMeters * Math.cos(angle) * lngPerMeter
            val dy = radiusMeters * Math.sin(angle) * latPerMeter
            vertices.add(Point.fromLngLat(centerLng + dx, centerLat + dy))
        }
        return vertices
    }
    
    private fun haversineDistance(
        lat1: Double, lng1: Double,
        lat2: Double, lng2: Double
    ): Double {
        val R = 6371000.0
        val dLat = Math.toRadians(lat2 - lat1)
        val dLng = Math.toRadians(lng2 - lng1)
        val a = Math.sin(dLat / 2).pow(2) +
            Math.cos(Math.toRadians(lat1)) * Math.cos(Math.toRadians(lat2)) *
            Math.sin(dLng / 2).pow(2)
        return R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a))
    }
}

class WebSocketClient @Inject constructor(
    private val okHttpClient: OkHttpClient,
    @Named("wsUrl") private val wsUrl: String
) {
    private var webSocket: WebSocket? = null
    private var reconnectAttempts = 0
    private val maxReconnectAttempts = 10
    private val _events = MutableSharedFlow<WebSocketEvent>(
        extraBufferCapacity = 64,
        onBufferOverflow = BufferOverflow.DROP_OLDEST
    )
    val events: SharedFlow<WebSocketEvent> = _events.asSharedFlow()

    private val scope = CoroutineScope(SupervisorJob() + Dispatchers.IO)

    fun connect(authToken: String) {
        if (webSocket != null) return

        val request = Request.Builder()
            .url("$wsUrl?token=$authToken")
            .build()

        webSocket = okHttpClient.newWebSocket(request, object : WebSocketListener() {
            override fun onOpen(webSocket: WebSocket, response: Response) {
                reconnectAttempts = 0
                _events.tryEmit(WebSocketEvent.Connected)
                startHeartbeat()
            }

            override fun onMessage(webSocket: WebSocket, text: String) {
                try {
                    val json = JSONObject(text)
                    when (json.getString("type")) {
                        "demand-update" -> handleDemandUpdate(json)
                        "pulse-new" -> handleNewPulse(json)
                        "pulse-expired" -> handlePulseExpired(json)
                        "recommendation-update" -> handleRecommendation(json)
                        "heartbeat" -> { /* connection alive */ }
                    }
                } catch (e: Exception) {
                    Log.e("WebSocketClient", "Failed to parse message", e)
                }
            }

            override fun onClosed(webSocket: WebSocket, code: Int, reason: String) {
                _events.tryEmit(WebSocketEvent.Disconnected(reason))
                attemptReconnect(authToken)
            }

            override fun onFailure(webSocket: WebSocket, t: Throwable, response: Response?) {
                _events.tryEmit(WebSocketEvent.Error(t))
                attemptReconnect(authToken)
            }
        })
    }

    fun sendDriverLocation(lat: Double, lng: Double, heading: Float, speed: Float) {
        val payload = JSONObject().apply {
            put("type", "driver-location")
            put("lat", lat)
            put("lng", lng)
            put("heading", heading)
            put("speed", speed)
        }
        webSocket?.send(payload.toString())
    }

    fun markRecommendationOutcome(followed: Boolean, recommendationId: String) {
        val payload = JSONObject().apply {
            put("type", "recommendation-feedback")
            put("followed", followed)
            put("id", recommendationId)
        }
        webSocket?.send(payload.toString())
    }

    fun disconnect() {
        webSocket?.close(1000, "Client disconnected")
        webSocket = null
        scope.cancel()
    }

    private fun attemptReconnect(authToken: String) {
        if (reconnectAttempts >= maxReconnectAttempts) {
            _events.tryEmit(WebSocketEvent.MaxReconnectReached)
            return
        }
        reconnectAttempts++
        scope.launch {
            delay(3000L * reconnectAttempts)
            connect(authToken)
        }
    }

    private fun startHeartbeat() {
        scope.launch {
            while (isActive) {
                delay(30000)
                webSocket?.send(JSONObject().put("type", "ping").toString())
            }
        }
    }

    private fun handleDemandUpdate(json: JSONObject) {
        val segmentsArray = json.getJSONArray("segments")
        val pulsesArray = json.getJSONArray("pulses")
        
        val segments = (0 until segmentsArray.length()).map { i ->
            parseSegment(segmentsArray.getJSONObject(i))
        }
        val pulses = (0 until pulsesArray.length()).map { i ->
            parsePulse(pulsesArray.getJSONObject(i))
        }
        
        _events.tryEmit(WebSocketEvent.DemandUpdate(segments, pulses))
    }

    private fun handleNewPulse(json: JSONObject) {
        val pulse = parsePulse(json.getJSONObject("pulse"))
        _events.tryEmit(WebSocketEvent.PulseCreated(pulse))
    }

    private fun handlePulseExpired(json: JSONObject) {
        val pulseId = json.getString("pulseId")
        _events.tryEmit(WebSocketEvent.PulseExpired(pulseId))
    }

    private fun handleRecommendation(json: JSONObject) {
        val rec = parseRecommendation(json.getJSONObject("recommendation"))
        _events.tryEmit(WebSocketEvent.RecommendationUpdated(rec))
    }

    // Parse helpers omitted for brevity
}

sealed class WebSocketEvent {
    object Connected : WebSocketEvent()
    data class Disconnected(val reason: String) : WebSocketEvent()
    data class Error(val throwable: Throwable) : WebSocketEvent()
    object MaxReconnectReached : WebSocketEvent()
    data class DemandUpdate(
        val segments: List<RoadSegmentHeat>,
        val pulses: List<DemandPulse>
    ) : WebSocketEvent()
    data class PulseCreated(val pulse: DemandPulse) : WebSocketEvent()
    data class PulseExpired(val pulseId: String) : WebSocketEvent()
    data class RecommendationUpdated(
        val recommendation: CopilotRecommendation
    ) : WebSocketEvent()
}
