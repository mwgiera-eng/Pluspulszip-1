Unified Design System — PlusPuls
Color Philosophy
One accent. Color serves data, never decorates. The palette is intentionally restrained — 90% of the interface is monochrome dark, and teal appears only where it carries meaning: active states, money values, and primary actions. Amber and coral are even rarer, reserved exclusively for caution and danger signals.

Design principle
info
If removing the color wouldn't change the meaning, the color shouldn't be there. Teal = go/money/active. Amber = caution. Coral = danger. Everything else is grayscale.

object PlusPulsEasing {
    // Primary — used for 80% of animations
    val FastOutSlowIn = CubicBezierEasing(0.4f, 0.0f, 0.2f, 1.0f)
    
    // Entering elements — slight acceleration
    val LinearOutSlowIn = CubicBezierEasing(0.0f, 0.0f, 0.2f, 0.1f)
    
    // Exiting elements — slight deceleration
    val FastOutLinearIn = CubicBezierEasing(0.4f, 0.0f, 1.0f, 1.0f)
    
    // Never use bounce springs — this app is confident, not playful
}

Compose Animation Patterns (Reusable)
// 1. Screen enter animation — applied to root composable of each screen
@Composable
fun AnimatedScreenEnter(content: @Composable () -> Unit) {
    var visible by remember { mutableStateOf(false) }
    LaunchedEffect(Unit) { visible = true }
    
    AnimatedVisibility(
        visible = visible,
        enter = fadeIn(
            animationSpec = tween(
                durationMillis = 220,
                easing = PlusPulsEasing.FastOutSlowIn
            )
        ) + slideInVertically(
            initialOffsetY = { 24 },
            animationSpec = tween(
                durationMillis = 220,
                easing = PlusPulsEasing.FastOutSlowIn
            )
        )
    ) {
        content()
    }
}

// 3. Button press scale — modifier extension
fun Modifier.pressScale(
    pressed: Boolean,
    scaleDown: Float = 0.97f
): Modifier = this.then(
    graphicsLayer {
        scaleX = animateFloatAsState(
            targetValue = if (pressed) scaleDown else 1f,
            animationSpec = tween(
                durationMillis = if (pressed) 100 else 180,
                easing = if (pressed) PlusPulsEasing.LinearOutSlowIn 
                         else PlusPulsEasing.FastOutSlowIn
            ),
            label = "press_scale"
        ).value
        scaleY = scaleX
    }
)

// 5. Pulse ring for driver location on map
@Composable
fun DriverPulseMarker(modifier: Modifier = Modifier) {
    val transition = rememberInfiniteTransition(label = "pulse")
    
    val ringScale by transition.animateFloat(
        initialValue = 1f,
        targetValue = 2.5f,
        animationSpec = infiniteRepeatable(
            animation = tween(2000, easing = LinearEasing),
            repeatMode = RepeatMode.Restart
        ),
        label = "ring_scale"
    )
    
    val ringAlpha by transition.animateFloat(
        initialValue = 0.6f,
        targetValue = 0f,
        animationSpec = infiniteRepeatable(
            animation = tween(2000, easing = LinearEasing),
            repeatMode = RepeatMode.Restart
        ),
        label = "ring_alpha"
    )
    
    Box(modifier = modifier.size(10.dp), contentAlignment = Alignment.Center) {
        Canvas(Modifier.matchParentSize()) {
            // Expanding ring
            drawCircle(
                color = Color(0xFF2EE6A6).copy(alpha = ringAlpha),
                radius = size.minDimension / 2 * ringScale,
                style = Stroke(width = 1.dp.toPx())
            )
            // Solid center dot
            drawCircle(
                color = Color(0xFF2EE6A6),
                radius = size.minDimension / 4
            )
        }
    }
}

// 7. Shimmer loading skeleton
@Composable
fun PlusPulsShimmer(
    modifier: Modifier = Modifier,
    shape: Shape = RoundedCornerShape(12.dp)
) {
    val transition = rememberInfiniteTransition(label = "shimmer")
    val translateX by transition.animateFloat(
        initialValue = -300f,
        targetValue = 300f,
        animationSpec = infiniteRepeatable(
            animation = tween(1400, easing = LinearEasing),
            repeatMode = RepeatMode.Restart
        ),
        label = "shimmer_x"
    )
    
    Box(
        modifier = modifier
            .clip(shape)
            .background(Color(0xFF1A1F2E))
            .drawWithContent {
                drawContent()
                drawRect(
                    brush = Brush.linearGradient(
                        colors = listOf(
                            Color.Transparent,
                            Color.White.copy(alpha = 0.04f),
                            Color.Transparent
                        ),
                        start = Offset(translateX, 0f),
                        end = Offset(translateX + 200f, size.height)
                    )
                )
            }
    )
}

// 10. Snackbar slide-in
@Composable
fun PlusPulsSnackbar(
    message: String,
    modifier: Modifier = Modifier
) {
    var visible by remember { mutableStateOf(false) }
    LaunchedEffect(message) { 
        delay(50)
        visible = true 
    }

        enter = slideInVertically(

            shape = RoundedCornerShape(14.dp),
            modifier = modifier.padding(16.dp)
        ) {
            Text(message, fontSize = 13.sp)
        }
    }
}

Animation Choreography Rules
These rules govern how multiple animations interact on a single screen to produce the "fluent" feeling:

Rule 1: Stagger, don't sync. When multiple elements animate in on screen enter, stagger their start delays by 40ms each. The topmost element starts at 0ms, the next at 40ms, then 80ms, etc. No two elements should reach their final state simultaneously — that looks robotic.

Rule 2: Enter before exit completes. When replacing content (tab switch, recommendation replacement), start the incoming element's entrance at 60% through the outgoing element's exit. This creates overlap that feels continuous rather than sequential.

Rule 3: Only one thing pulses. At any given moment, at most one looping animation runs on screen — the driver location pulse. If a snackbar appears, the pulse pauses for its duration. Never have two infinite-loop animations running simultaneously.

Rule 4: Map transitions are slower. Map camera movements and hex color transitions use 350–400ms durations, while UI chrome uses 180–220ms. The map is heavier visually and needs more time to settle. If UI and map animate simultaneously, start UI at 0ms and map at 50ms so UI settles first and the map finishes last.

Rule 5: Respect reduced motion. Check LocalAccessibilityManager.current.isReduceMotionEnabled and if true, replace all animation specs with tween(0) or snap(). Loop animations (pulse, shimmer) become static.

@Composable
fun rememberPlusPulsSpec(
    duration: Int
): AnimationSpec<Float> {
    val reduceMotion = LocalAccessibilityManager.current
        ?.isReduceMotionEnabled ?: false
    return if (reduceMotion) {
        snap(delayMillis = 0)
    } else {
        tween(
            durationMillis = duration,
            easing = PlusPulsEasing.FastOutSlowIn
        )
    }
}

Motion philosophy
info
Every animation above 250ms is a choreography opportunity — stagger elements, overlap enter/exit, and let the map settle slower than the chrome. The app should feel like it anticipates the user's next glance: the eye lands where the animation just finished settling.
